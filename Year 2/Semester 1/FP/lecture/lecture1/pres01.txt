* What is functional programming?
(from Graham Hutton)

Functional programming is style of programming in which the basic
method of computation is the application of functions to arguments;

A functional language is one that supports and encourages the
functional style.

** Java example
int total = 0;
for (int i = 1; i <= 10; i++) {
  total = total + i;
}

Method of computation: variable assignment.
** Haskell example
sum [1..10]

Method of computation: function application.

* Taste of Haskell 3
#+BEGIN_EXAMPLE
f []     = [] 
f (x:xs) = f as ++ [x] ++ f bs
  where
    as = [a | a <- xs, a <= x]
    bs = [b | b <- xs, x <  b]
#+END_EXAMPLE
** Same in C++
template <typename T>
void qsort (T *result, T *list, int n)
{
    if (n == 0) return;
    T *smallerList, *largerList;
    smallerList = new T[n];
    largerList = new T[n];      
    T pivot = list[0];
    int numSmaller=0, numLarger=0;      
    for (int i = 1; i < n; i++)
        if (list[i] < pivot)
            smallerList[numSmaller++] = list[i];
        else 
            largerList[numLarger++] = list[i];
    
    qsort(smallerList,smallerList,numSmaller); 
    qsort(largerList,largerList,numLarger);
    
    int pos = 0;        
    for ( int i = 0; i < numSmaller; i++)
        result[pos++] = smallerList[i];
    
    result[pos++] = pivot;
    
    for ( int i = 0; i < numLarger; i++)
        result[pos++] = largerList[i];
    
    delete [] smallerList;
    delete [] largerList;
};
